cmake_minimum_required(VERSION 3.20)
project(box3_preview C)
set(CMAKE_C_STANDARD 11)
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h" CACHE PATH "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
get_filename_component(DEFAULT_BOX_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../box3-zephyr" ABSOLUTE)
set(BOX3_ZEPHYR_DIR "${DEFAULT_BOX_DIR}" CACHE PATH "BOX-3 Zephyr workspace root")
set(LVGL_DIR "${BOX3_ZEPHYR_DIR}/workspace/modules/lib/gui/lvgl" CACHE PATH "Pinned LVGL source directory")
add_subdirectory("${LVGL_DIR}" lvgl)
add_executable(preview main.c "${BOX3_ZEPHYR_DIR}/app/src/ui.c" "${BOX3_ZEPHYR_DIR}/app/src/forecast.c")
target_include_directories(preview PRIVATE "${BOX3_ZEPHYR_DIR}/app/src")
target_link_libraries(preview PRIVATE lvgl)
